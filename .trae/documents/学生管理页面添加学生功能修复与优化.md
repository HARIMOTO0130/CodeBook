2. **添加搜索学生方法**：
   ```javascript
   methods: {
     // 现有方法...
     async searchStudents() {
       if (!this.searchExistingStudent.trim()) return
       
       this.searching = true
       try {
         const response = await studentApi.getStudents({
           search: this.searchExistingStudent
         })
         
         // 处理不同的响应格式
         let data = response.data
         let results = []
         
         if (Array.isArray(data)) {
           results = data
         } else if (data && Array.isArray(data.results)) {
           results = data.results
         } else if (data && Array.isArray(data.data)) {
           results = data.data
         } else {
           results = []
         }
         
         this.searchResults = results.map(student => ({
           id: student.id,
           name: student.student_name || student.username,
           studentId: student.student_no || '',
           currentClass: student.class_name || '未分配班级'
         }))
       } catch (error) {
         console.error('搜索学生失败:', error)
         this.showMessage('搜索学生失败', 'error')
       } finally {
         this.searching = false
       }
     },
     
     toggleStudentSelection(studentId) {
       const index = this.selectedStudents.indexOf(studentId)
       if (index > -1) {
         this.selectedStudents.splice(index, 1)
       } else {
         this.selectedStudents.push(studentId)
       }
     },
     
     selectAllStudents() {
       if (this.selectedStudents.length === this.searchResults.length) {
         this.selectedStudents = []
       } else {
         this.selectedStudents = this.searchResults.map(s => s.id)
       }
     },
     
     async assignStudentsToClass() {
       if (this.selectedStudents.length === 0) {
         this.showMessage('请选择要分配的学生', 'warning')
         return
       }
       
       if (!this.assigningClassId) {
         this.showMessage('请选择班级', 'warning')
         return
       }
       
       this.assigning = true
       try {
         // 调用API将选中的学生分配到指定班级
         // 假设API支持批量分配
         await studentApi.assignStudentsToClass({
           student_ids: this.selectedStudents,
           class_id: this.assigningClassId
         })
         
         this.showMessage(`成功分配${this.selectedStudents.length}名学生到班级`, 'success')
         this.closeModal()
         await this.loadStudents() // 刷新学生列表
       } catch (error) {
         console.error('分配学生失败:', error)
         this.showMessage('分配学生失败', 'error')
       } finally {
         this.assigning = false
       }
     },
     
     showMessage(text, type) {
       this.message = text
       this.messageType = type
       setTimeout(() => {
         this.message = ''
         this.messageType = ''
       }, 3000)
     }
   }
   ```
